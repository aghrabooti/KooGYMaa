### چطور محاسبه می‌شود؟

```ts
function calculateMonthsAsStudent(startedAt: Date): number {
  const now = new Date();
  const years = now.getFullYear() - startedAt.getFullYear();
  const months = now.getMonth() - startedAt.getMonth();
  return years * 12 + months;
}

// برای نیما:
// startedAt = 2026-03-16
// now = 2026-09-16
// months = 6
```
